cmake_minimum_required(VERSION 3.20)
project(FilmDevelopWindowsEngine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32)
  message(FATAL_ERROR "此引擎使用 Windows WIC，請使用 Windows 工具鏈")
endif()
set(CORE "${CMAKE_CURRENT_SOURCE_DIR}/../../experiments/PhotoCoreCpp")
add_executable(filmdevelop-engine main.cpp codec.cpp neural.cpp repair.cpp inference.cpp vision.cpp
  "${CORE}/src/core.cpp" "${CORE}/src/film_data.cpp" "${CORE}/src/film_image.cpp"
  "${CORE}/src/film_development.cpp" "${CORE}/src/film_spectral.cpp" "${CORE}/src/film_scanner.cpp" "${CORE}/src/film_optics.cpp" "${CORE}/src/film_digital.cpp")
target_include_directories(filmdevelop-engine PRIVATE ../cpp "${CORE}/include" "${CORE}/src" ../../Vendor/PhotoCompute/src)
target_include_directories(filmdevelop-engine PRIVATE ../../Vendor/PhotoRAW/src)
if(NOT EXISTS "${PHOTO_RAW_LIBRARY}")
  message(FATAL_ERROR "需先執行 build-raw-windows.py，並設定 PHOTO_RAW_LIBRARY")
endif()
target_include_directories(filmdevelop-engine SYSTEM PRIVATE "${CORE}/third_party")
target_compile_definitions(filmdevelop-engine PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(filmdevelop-engine PRIVATE "${PHOTO_RAW_LIBRARY}" windowscodecs ole32 oleaut32 uuid crypt32 ws2_32)
if(NOT EXISTS "${PHOTO_WEBP_LIBRARY}")
  message(FATAL_ERROR "需先執行 build-webp-windows.py，並設定 PHOTO_WEBP_LIBRARY")
endif()
target_include_directories(filmdevelop-engine PRIVATE ../../aiTest/ThirdParty/stable-diffusion.cpp/thirdparty/libwebp/src)
target_link_libraries(filmdevelop-engine PRIVATE "${PHOTO_WEBP_LIBRARY}")
if(NOT EXISTS "${PHOTO_NEURAL_DIR}/include/onnxruntime_cxx_api.h")
  message(FATAL_ERROR "需先執行 prepare-neural-windows.py，並設定 PHOTO_NEURAL_DIR")
endif()
target_include_directories(filmdevelop-engine SYSTEM PRIVATE "${PHOTO_NEURAL_DIR}/include")
target_link_libraries(filmdevelop-engine PRIVATE dxgi)
if(MSVC)
  target_compile_options(filmdevelop-engine PRIVATE /W4 /utf-8 /fp:precise)
else()
  target_compile_options(filmdevelop-engine PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()

set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../aiTest2/ThirdParty/llama.cpp")
if(NOT EXISTS "${PHOTO_LLAMA_DIR}/src/libllama.dll.a")
  message(FATAL_ERROR "需先執行 build-llama-windows.sh，並設定 PHOTO_LLAMA_DIR")
endif()
target_include_directories(filmdevelop-engine SYSTEM PRIVATE "${LLAMA_SOURCE}/include" "${LLAMA_SOURCE}/ggml/include" "${LLAMA_SOURCE}/tools/mtmd")
target_link_libraries(filmdevelop-engine PRIVATE "${PHOTO_LLAMA_DIR}/tools/mtmd/libmtmd.dll.a" "${PHOTO_LLAMA_DIR}/src/libllama.dll.a" "${PHOTO_LLAMA_DIR}/ggml/src/libggml.dll.a" "${PHOTO_LLAMA_DIR}/ggml/src/libggml-base.dll.a")
