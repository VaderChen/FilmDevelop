cmake_minimum_required(VERSION 3.20)
project(FilmDevelopEngineContract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(filmdevelop-contract-smoke verify_contract.cpp)
target_include_directories(filmdevelop-contract-smoke SYSTEM PRIVATE ../../experiments/PhotoCoreCpp/third_party)
if(MSVC)
  target_compile_options(filmdevelop-contract-smoke PRIVATE /W4 /utf-8)
else()
  target_compile_options(filmdevelop-contract-smoke PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME generated_contract COMMAND filmdevelop-contract-smoke --self-test)
